cmake_minimum_required(VERSION 3.18)
project(farm_control CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Device config for this machine: API/src/config/config_<DEVICE_CONFIG>.cpp
#   cmake -B build -DDEVICE_CONFIG=uppertank
# No default on purpose: building the wrong config on a RevPi drives the wrong outputs.
set(DEVICE_CONFIG "" CACHE STRING "Device config to build (config_<name>.cpp in API/src/config)")

set(config_dir ${CMAKE_CURRENT_SOURCE_DIR}/API/src/config)
file(GLOB config_files RELATIVE ${config_dir} ${config_dir}/config_*.cpp)
string(REPLACE "config_" "" available_configs "${config_files}")
string(REPLACE ".cpp" "" available_configs "${available_configs}")
set_property(CACHE DEVICE_CONFIG PROPERTY STRINGS ${available_configs})

set(config_file ${config_dir}/config_${DEVICE_CONFIG}.cpp)
if(DEVICE_CONFIG STREQUAL "" OR NOT EXISTS ${config_file})
    message(FATAL_ERROR "Set -DDEVICE_CONFIG=<name>, one of: ${available_configs}")
endif()
message(STATUS "Device config: ${DEVICE_CONFIG}")

# API: everything under API/src except the configs, plus the chosen config
file(GLOB_RECURSE api_sources CONFIGURE_DEPENDS API/src/*.cpp)
list(FILTER api_sources EXCLUDE REGEX "/API/src/config/")
add_library(farm_api STATIC ${api_sources} ${config_file})
target_include_directories(farm_api
    PUBLIC API/include    # what controllers include
    PRIVATE API/src)
target_compile_options(farm_api PRIVATE -Wall -Wextra)

# Web UI password, hardcoded at build time from the first line of webui_password.txt
# (repo root, git-ignored). Missing or empty: the web UI refuses to start.
# Editing the file re-runs cmake on the next build.
set(password_file ${CMAKE_CURRENT_SOURCE_DIR}/webui_password.txt)
set(WEB_UI_PASSWORD "")
if(EXISTS ${password_file})
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${password_file})
    file(READ ${password_file} WEB_UI_PASSWORD)
    string(REGEX REPLACE "[\r\n].*$" "" WEB_UI_PASSWORD "${WEB_UI_PASSWORD}")
    string(STRIP "${WEB_UI_PASSWORD}" WEB_UI_PASSWORD)
endif()
if(WEB_UI_PASSWORD STREQUAL "")
    message(WARNING "No password in ${password_file}: the web UI will not start")
endif()
string(REPLACE "\\" "\\\\" WEB_UI_PASSWORD "${WEB_UI_PASSWORD}")
string(REPLACE "\"" "\\\"" WEB_UI_PASSWORD "${WEB_UI_PASSWORD}")
file(CONFIGURE OUTPUT ${CMAKE_BINARY_DIR}/generated/web_ui_password.h
     CONTENT "#pragma once\n// Generated by CMake from webui_password.txt, do not edit or commit\ninline constexpr char WEB_UI_PASSWORD[] = \"@WEB_UI_PASSWORD@\";\n"
     @ONLY)
target_include_directories(farm_api PRIVATE ${CMAKE_BINARY_DIR}/generated)
find_package(Threads REQUIRED)
target_link_libraries(farm_api PUBLIC Threads::Threads)

# Notifications: ntfy push notifications (notifications/src), host PC only: the RevPi
# builds never compile them. Topic and server are hardcoded at build time from
# ntfy_topic.txt (repo root, git-ignored): line 1 = topic, optional line 2 = server
# (default https://ntfy.sh). Missing: notifications are off.
if(DEVICE_CONFIG STREQUAL "host_pc")
    set(ntfy_file ${CMAKE_CURRENT_SOURCE_DIR}/ntfy_topic.txt)
    set(NTFY_TOPIC "")
    set(NTFY_SERVER "https://ntfy.sh")
    if(EXISTS ${ntfy_file})
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${ntfy_file})
        file(STRINGS ${ntfy_file} ntfy_lines)
        list(LENGTH ntfy_lines ntfy_line_count)
        if(ntfy_line_count GREATER 0)
            list(GET ntfy_lines 0 NTFY_TOPIC)
            string(STRIP "${NTFY_TOPIC}" NTFY_TOPIC)
        endif()
        if(ntfy_line_count GREATER 1)
            list(GET ntfy_lines 1 ntfy_server_line)
            string(STRIP "${ntfy_server_line}" ntfy_server_line)
            if(NOT ntfy_server_line STREQUAL "")
                set(NTFY_SERVER "${ntfy_server_line}")
            endif()
        endif()
    endif()
    if(NTFY_TOPIC STREQUAL "")
        message(WARNING "No topic in ${ntfy_file}: notifications are off")
    elseif(NOT NTFY_TOPIC MATCHES "^[A-Za-z0-9_-]+$")
        message(FATAL_ERROR "ntfy topic may only contain letters, digits, _ and -")
    endif()
    string(REPLACE "\"" "" NTFY_SERVER "${NTFY_SERVER}")
    string(REPLACE "\\" "" NTFY_SERVER "${NTFY_SERVER}")
    file(CONFIGURE OUTPUT ${CMAKE_BINARY_DIR}/generated/ntfy_config.h
         CONTENT "#pragma once\n// Generated by CMake from ntfy_topic.txt, do not edit or commit\ninline constexpr char NTFY_TOPIC[] = \"@NTFY_TOPIC@\";\ninline constexpr char NTFY_SERVER[] = \"@NTFY_SERVER@\";\n"
         @ONLY)
    add_library(notifications STATIC notifications/src/ntfy.cpp)
    target_include_directories(notifications PUBLIC notifications/src ${CMAKE_BINARY_DIR}/generated)
    target_compile_options(notifications PRIVATE -Wall -Wextra)
    target_link_libraries(notifications PUBLIC Threads::Threads)
    add_executable(ntfy_send notifications/src/ntfy_send.cpp)
    target_compile_options(ntfy_send PRIVATE -Wall -Wextra)
    target_link_libraries(ntfy_send PRIVATE notifications)
endif()

# User space: one executable per user_space/src/*.cpp, named after the file.
# main_<config>.cpp is only built with that config; other files with every config.
file(GLOB user_sources CONFIGURE_DEPENDS user_space/src/*.cpp)
foreach(source ${user_sources})
    get_filename_component(name ${source} NAME_WE)
    if(name MATCHES "^main_(.+)$" AND NOT CMAKE_MATCH_1 STREQUAL DEVICE_CONFIG)
        continue()
    endif()
    add_executable(${name} ${source})
    target_compile_options(${name} PRIVATE -Wall -Wextra)
    target_link_libraries(${name} PRIVATE farm_api)
endforeach()
if(TARGET main_host_pc)
    target_link_libraries(main_host_pc PRIVATE notifications)
endif()
