cmake_minimum_required(VERSION 3.18)
project(farm_control CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# compile_commands.json in the build folder, for the editor (.vscode/c_cpp_properties.json)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Device config for this machine: API/src/config/config_<DEVICE_CONFIG>.cpp
#   cmake -B build -DDEVICE_CONFIG=uppertank
# No default on purpose: building the wrong config on a RevPi drives the wrong outputs.
set(DEVICE_CONFIG "" CACHE STRING "Device config to build (config_<name>.cpp in API/src/config)")

set(config_dir ${CMAKE_CURRENT_SOURCE_DIR}/API/src/config)
file(GLOB config_files RELATIVE ${config_dir} ${config_dir}/config_*.cpp)
string(REPLACE "config_" "" available_configs "${config_files}")
string(REPLACE ".cpp" "" available_configs "${available_configs}")
set_property(CACHE DEVICE_CONFIG PROPERTY STRINGS ${available_configs})

set(config_file ${config_dir}/config_${DEVICE_CONFIG}.cpp)
if(DEVICE_CONFIG STREQUAL "" OR NOT EXISTS ${config_file})
    message(FATAL_ERROR "Set -DDEVICE_CONFIG=<name>, one of: ${available_configs}")
endif()
message(STATUS "Device config: ${DEVICE_CONFIG}")

find_package(Threads REQUIRED)

# Comms (comms/src): line protocol between farm machines, independent of the API.
# line_server (port 47100, routes requests to handlers by name, client watchdog)
# and remote_link (client side). Used by the API's device server and remote devices.
add_library(comms STATIC comms/src/line_server.cpp comms/src/remote_link.cpp)
target_include_directories(comms PUBLIC comms/src)
target_compile_options(comms PRIVATE -Wall -Wextra)
target_link_libraries(comms PUBLIC Threads::Threads)

# API: everything under API/src except the configs, plus the chosen config
file(GLOB_RECURSE api_sources CONFIGURE_DEPENDS API/src/*.cpp)
list(FILTER api_sources EXCLUDE REGEX "/API/src/config/")
# WiseConn cloud devices: host PC only (needs internet and the API key)
if(NOT DEVICE_CONFIG STREQUAL "host_pc")
    list(FILTER api_sources EXCLUDE REGEX "/API/src/devices/wiseconn_devices/")
endif()
add_library(farm_api STATIC ${api_sources} ${config_file})
target_include_directories(farm_api
    PUBLIC API/include    # what controllers include
    PRIVATE API/src)
target_compile_options(farm_api PRIVATE -Wall -Wextra)

# WiseConn keys: wiseconn_keys.txt (repo root, git-ignored: API key, farm, zone and
# measure ids) built in as text at build time, parsed by config_host_pc at startup.
# Missing: the WiseConn devices are off. Editing the file re-runs cmake.
if(DEVICE_CONFIG STREQUAL "host_pc")
    set(wiseconn_file ${CMAKE_CURRENT_SOURCE_DIR}/wiseconn_keys.txt)
    set(WISECONN_KEYS_TEXT "")
    if(EXISTS ${wiseconn_file})
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${wiseconn_file})
        file(READ ${wiseconn_file} WISECONN_KEYS_TEXT)
    else()
        message(WARNING "No ${wiseconn_file}: WiseConn devices are off")
    endif()
    string(REPLACE "\\" "\\\\" WISECONN_KEYS_TEXT "${WISECONN_KEYS_TEXT}")
    string(REPLACE "\"" "\\\"" WISECONN_KEYS_TEXT "${WISECONN_KEYS_TEXT}")
    string(REPLACE "\r" "" WISECONN_KEYS_TEXT "${WISECONN_KEYS_TEXT}")
    string(REPLACE "\n" "\\n\"\n    \"" WISECONN_KEYS_TEXT "${WISECONN_KEYS_TEXT}")
    # Own folder so only the API sees the key
    file(CONFIGURE OUTPUT ${CMAKE_BINARY_DIR}/generated/wiseconn/wiseconn_keys_text.h
         CONTENT "#pragma once\n// Generated by CMake from wiseconn_keys.txt, do not edit or commit\ninline constexpr char WISECONN_KEYS_TEXT[] =\n    \"@WISECONN_KEYS_TEXT@\";\n"
         @ONLY)
    target_include_directories(farm_api PRIVATE ${CMAKE_BINARY_DIR}/generated/wiseconn)
endif()

target_link_libraries(farm_api PUBLIC comms)   # device server, remote devices

# Web UI (webUI/src): HTTP page over the public device interfaces, host PC only: the
# RevPi builds never compile it. Password hardcoded at build time from the first line
# of webui_password.txt (repo root, git-ignored); missing or empty: start() fails.
# Editing the file re-runs cmake on the next build.
if(DEVICE_CONFIG STREQUAL "host_pc")
    set(password_file ${CMAKE_CURRENT_SOURCE_DIR}/webui_password.txt)
    set(WEB_UI_PASSWORD "")
    if(EXISTS ${password_file})
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${password_file})
        file(READ ${password_file} WEB_UI_PASSWORD)
        string(REGEX REPLACE "[\r\n].*$" "" WEB_UI_PASSWORD "${WEB_UI_PASSWORD}")
        string(STRIP "${WEB_UI_PASSWORD}" WEB_UI_PASSWORD)
    endif()
    if(WEB_UI_PASSWORD STREQUAL "")
        message(WARNING "No password in ${password_file}: the web UI will not start")
    endif()
    string(REPLACE "\\" "\\\\" WEB_UI_PASSWORD "${WEB_UI_PASSWORD}")
    string(REPLACE "\"" "\\\"" WEB_UI_PASSWORD "${WEB_UI_PASSWORD}")
    # Own folder so only web_ui.cpp sees the password
    file(CONFIGURE OUTPUT ${CMAKE_BINARY_DIR}/generated/web_ui/web_ui_password.h
         CONTENT "#pragma once\n// Generated by CMake from webui_password.txt, do not edit or commit\ninline constexpr char WEB_UI_PASSWORD[] = \"@WEB_UI_PASSWORD@\";\n"
         @ONLY)
    add_library(web_ui STATIC webUI/src/web_ui.cpp)
    target_include_directories(web_ui
        PUBLIC webUI/src
        PRIVATE ${CMAKE_BINARY_DIR}/generated/web_ui)
    target_compile_options(web_ui PRIVATE -Wall -Wextra)
    target_link_libraries(web_ui PUBLIC farm_api)   # devices.h
endif()

# Notifications: ntfy push notifications (notifications/src), host PC only: the RevPi
# builds never compile them. Topic and server are hardcoded at build time from
# ntfy_topic.txt (repo root, git-ignored): line 1 = topic, optional line 2 = server
# (default https://ntfy.sh). Missing: notifications are off.
if(DEVICE_CONFIG STREQUAL "host_pc")
    set(ntfy_file ${CMAKE_CURRENT_SOURCE_DIR}/ntfy_topic.txt)
    set(NTFY_TOPIC "")
    set(NTFY_SERVER "https://ntfy.sh")
    if(EXISTS ${ntfy_file})
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${ntfy_file})
        file(STRINGS ${ntfy_file} ntfy_lines)
        list(LENGTH ntfy_lines ntfy_line_count)
        if(ntfy_line_count GREATER 0)
            list(GET ntfy_lines 0 NTFY_TOPIC)
            string(STRIP "${NTFY_TOPIC}" NTFY_TOPIC)
        endif()
        if(ntfy_line_count GREATER 1)
            list(GET ntfy_lines 1 ntfy_server_line)
            string(STRIP "${ntfy_server_line}" ntfy_server_line)
            if(NOT ntfy_server_line STREQUAL "")
                set(NTFY_SERVER "${ntfy_server_line}")
            endif()
        endif()
    endif()
    if(NTFY_TOPIC STREQUAL "")
        message(WARNING "No topic in ${ntfy_file}: notifications are off")
    elseif(NOT NTFY_TOPIC MATCHES "^[A-Za-z0-9_-]+$")
        message(FATAL_ERROR "ntfy topic may only contain letters, digits, _ and -")
    endif()
    string(REPLACE "\"" "" NTFY_SERVER "${NTFY_SERVER}")
    string(REPLACE "\\" "" NTFY_SERVER "${NTFY_SERVER}")
    file(CONFIGURE OUTPUT ${CMAKE_BINARY_DIR}/generated/ntfy_config.h
         CONTENT "#pragma once\n// Generated by CMake from ntfy_topic.txt, do not edit or commit\ninline constexpr char NTFY_TOPIC[] = \"@NTFY_TOPIC@\";\ninline constexpr char NTFY_SERVER[] = \"@NTFY_SERVER@\";\n"
         @ONLY)
    add_library(notifications STATIC notifications/src/ntfy.cpp)
    target_include_directories(notifications PUBLIC notifications/src ${CMAKE_BINARY_DIR}/generated)
    target_compile_options(notifications PRIVATE -Wall -Wextra)
    target_link_libraries(notifications PUBLIC Threads::Threads)
    add_executable(ntfy_send notifications/src/ntfy_send.cpp)
    target_compile_options(ntfy_send PRIVATE -Wall -Wextra)
    target_link_libraries(ntfy_send PRIVATE notifications)
endif()

# Scheduler (scheduler/): tasks run when their conditions are met (system clock),
# independent of the API. Single threaded: update() from the main loop.
add_library(scheduler STATIC scheduler/src/scheduler.cpp)
target_include_directories(scheduler PUBLIC scheduler/include)
target_compile_options(scheduler PRIVATE -Wall -Wextra)

# User space: one executable per user_space/src/*.cpp, named after the file.
# main_<config>.cpp is only built with that config; other files with every config.
file(GLOB user_sources CONFIGURE_DEPENDS user_space/src/*.cpp)
foreach(source ${user_sources})
    get_filename_component(name ${source} NAME_WE)
    if(name MATCHES "^main_(.+)$" AND NOT CMAKE_MATCH_1 STREQUAL DEVICE_CONFIG)
        continue()
    endif()
    add_executable(${name} ${source})
    target_compile_options(${name} PRIVATE -Wall -Wextra)
    target_link_libraries(${name} PRIVATE farm_api scheduler)   # every machine may own one scheduler
endforeach()
if(TARGET main_host_pc)
    target_link_libraries(main_host_pc PRIVATE web_ui notifications)
endif()

# RO controller (RO/): programs include RO/include/ro_config.h; the implementation is
# RO/src/config/ro_config_<DEVICE_CONFIG>.cpp, like the API configs. On the RO RevPi
# the ro state machine runs here and is served as "ro" (ro_server); on the host PC the
# same functions go to it through remote_ro, and the ro class is never built there, so
# no second controller can run. Other configs have no RO config and don't build it.
set(ro_config_file ${CMAKE_CURRENT_SOURCE_DIR}/RO/src/config/ro_config_${DEVICE_CONFIG}.cpp)
if(EXISTS ${ro_config_file})
    if(DEVICE_CONFIG STREQUAL "ro")
        # procedures (CIP cycles) drive the local ro on the scheduler
        set(ro_sources RO/src/ro.cpp RO/src/remote_ro/ro_server.cpp RO/src/procedures.cpp)
    else()
        set(ro_sources RO/src/remote_ro/remote_ro.cpp)
    endif()
    add_library(ro_control STATIC ${ro_sources} ${ro_config_file})
    target_include_directories(ro_control PUBLIC RO/include PRIVATE RO/src)
    target_compile_options(ro_control PRIVATE -Wall -Wextra)
    target_link_libraries(ro_control PUBLIC farm_api scheduler)   # config.h devices, comms; procedures
    target_link_libraries(main_${DEVICE_CONFIG} PRIVATE ro_control)
endif()
